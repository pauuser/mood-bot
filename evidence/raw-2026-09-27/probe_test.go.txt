// Пробные тесты исходной версии 81ed500. Они фиксируют текущее поведение, а не желаемое.

type fakeBot struct{ sent []string }

func (f *fakeBot) Request(string, string) error              { return nil }
func (f *fakeBot) SendMessage(_ int64, m string)              { f.sent = append(f.sent, m) }
func (f *fakeBot) SendQuestion(_ int64, _ string, _ []string) {}

func setup(t *testing.T) (*sql.DB, *messageProcessorImpl, *statisticsCronJob) {
	db, err := (&flags.SqliteFlags{Path: t.TempDir() + "/t.db"}).InitDB()
	if err != nil {
		t.Fatal(err)
	}
	l := zap.NewNop()
	qr := repository_impl.NewQuestionRepoSqliteImpl(db, l)
	ur := repository_impl.NewUserRepoSqliteImpl(db, l)
	fb := &fakeBot{}
	questions := map[string][]string{"Q?": {"1", "2", "3"}}
	mp := NewMessageProcessorImpl(l, fb, ur, qr, questions).(*messageProcessorImpl)
	sj := NewStatisticsCronJob(ur, qr, fb, l).(*statisticsCronJob)
	return db, mp, sj
}

func press(chat int64, question, data string) tgbotapi.Update {
	return tgbotapi.Update{CallbackQuery: &tgbotapi.CallbackQuery{ID: "1", Data: data,
		From:    &tgbotapi.User{ID: chat},
		Message: &tgbotapi.Message{Text: question, Chat: &tgbotapi.Chat{ID: chat, Type: "private"}}}}
}

// Сохранённый ответ не читается обратно: SELECT * возвращает 6 столбцов, Scan принимает 5.
func TestAnswerIsNotReadBack(t *testing.T) {
	_, mp, sj := setup(t)
	mp.Process(press(42, "Q?", "3"))

	all, _ := sj.questionRepository.GetAll()
	if len(all) != 1 {
		t.Fatalf("ожидалась 1 запись, получено %d", len(all))
	}
	a := all[0]
	t.Logf("прочитано: chat=%d question=%q answer=%q", a.FromChatId, a.QuestionText, a.Answer)
	if a.FromChatId != 0 || a.Answer != "" {
		t.Fatalf("ответ прочитан корректно — дефект не воспроизводится")
	}
	if n := len(sj.GetStatistics(all)); n != 0 {
		t.Fatalf("ожидалась пустая статистика, пользователей: %d", n)
	}
}

// Обработчик сохраняет значения, которых нет в конфигурации, и повторные ответы.
func TestCallbackStoresAnyData(t *testing.T) {
	db, mp, _ := setup(t)
	mp.Process(press(42, "Q?", "99999"))
	mp.Process(press(42, "Вопроса нет в конфиге", "<b>x</b>"))
	mp.Process(press(42, "Q?", "3"))
	mp.Process(press(42, "Q?", "3"))

	rows, err := db.Query("SELECT question_text, answer FROM questions WHERE from_chat_id = 42 ORDER BY id")
	if err != nil {
		t.Fatal(err)
	}
	defer rows.Close()
	var got [][2]string
	for rows.Next() {
		var q, a string
		rows.Scan(&q, &a)
		got = append(got, [2]string{q, a})
	}
	want := [][2]string{{"Q?", "99999"}, {"Вопроса нет в конфиге", "<b>x</b>"}, {"Q?", "3"}, {"Q?", "3"}}
	t.Logf("в БД: %q", got)
	if len(got) != len(want) {
		t.Fatalf("ожидалось %d записей, получено %d", len(want), len(got))
	}
	for i := range want {
		if got[i] != want[i] {
			t.Fatalf("запись %d: ожидалось %q, получено %q", i, want[i], got[i])
		}
	}
}
